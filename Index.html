    <script>

        let startTime = Date.now();

        let historyStack = [1];

        let finalStatusRecorded = "Completed";

        let recordedTimeSpent = "0.00";


        function goToStep(stepNumber) {

            historyStack.push(stepNumber);

            displayStep(stepNumber);

        }


        function prevStep() {

            if (historyStack.length > 1) {

                historyStack.pop();

                let previousStep = historyStack[historyStack.length - 1];

                displayStep(previousStep);

            }

        }


        function displayStep(stepNumber) {

            for (let i = 1; i <= 6; i++) {

                let el = document.getElementById('step-' + i);

                if (el) el.style.display = 'none';

            }

            let target = document.getElementById('step-' + stepNumber);

            if (target) target.style.display = 'block';


            if (stepNumber === 6) {

                let pb = document.getElementById('progress-container');

                let ti = document.getElementById('top-instruction');

                if (pb) pb.style.display = 'none';

                if (ti) ti.style.display = 'none';

            } else {

                let progress = Math.min((stepNumber / 5) * 100, 100);

                document.getElementById('progress-bar').style.width = progress + '%';

            }

        }


        function proceedFromStep3() {

            let sfSelected = document.getElementById('delivery-sf').checked;

            if (sfSelected) {

                goToStep(4); // 選擇順豐 -> 進入 PayMe $10 運費陷阱

            } else {

                goToStep(5); // 選擇校園自取 -> 進入完成登記頁面

            }

        }


        function exitSimulation(stepNumber) {

            triggerDebrief(`在第 ${stepNumber} 步離開`);

        }


        function triggerDebrief(status) {

            finalStatusRecorded = status;

            recordedTimeSpent = ((Date.now() - startTime) / 1000).toFixed(2);

            document.getElementById('stat-status').innerText = finalStatusRecorded;

            document.getElementById('stat-time').innerText = recordedTimeSpent;

            goToStep(6);

        }


        const _originalFinalFinish = function() {

            alert(`數據已記錄完成！\n\n- 狀態：${finalStatusRecorded}\n- 停留時間：${recordedTimeSpent} 秒\n\n請關閉此分頁，返回問卷填寫對應欄位。`);

        };


        const RETURN_URL = ""; 


        function returnToSurvey() {

            if (RETURN_URL) {

                window.location.href = RETURN_URL;

                return;

            }

            if (window.opener && !window.opener.closed) {

                try { window.opener.focus(); } catch (e) {}

                window.close();

                setTimeout(showManualReturnMessage, 400);

                return;

            }

            if (window.parent && window.parent !== window) {

                window.parent.postMessage({ action: 'returnToSurvey' }, '*');

                return;

            }

            if (window.history.length > 1) {

                window.history.back();

            } else {

                showManualReturnMessage();

            }

        }


        function showManualReturnMessage() {

            var overlay = document.createElement('div');

            overlay.style.cssText =

                'position:fixed;inset:0;background:rgba(255,255,255,0.97);z-index:9999;' +

                'display:flex;align-items:center;justify-content:center;padding:24px;text-align:center;' +

                'font-family:system-ui,-apple-system,"PingFang HK","Microsoft JhengHei",sans-serif;';

            overlay.innerHTML =

                '<div>' +

                    '<p style="font-size:18px;font-weight:700;color:#5b21b6;margin-bottom:12px;">感謝您的參與！</p>' +

                    '<p style="font-size:14px;color:#374151;line-height:1.8;">' +

                        '請關閉此分頁，返回原本的問卷頁面<br>繼續填寫餘下的問題。' +

                    '</p>' +

                '</div>';

            document.body.appendChild(overlay);

        }


        window.finalFinish = function () {

            _originalFinalFinish();

            returnToSurvey();

        };
